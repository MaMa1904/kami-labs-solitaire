import random


def card_to_string(card):
    """Convert card number (0-31) to readable format like 'Seven of Hearts'."""
    suits = ['Hearts', 'Diamonds', 'Clubs', 'Spades']
    ranks = ['Seven', 'Eight', 'Nine', 'Ten', 'Jack', 'Queen', 'King', 'Ace']
    
    suit_index = card // 8
    rank_index = card % 8
    
    return f"{ranks[rank_index]} of {suits[suit_index]}"


def get_suit(card):
    """Return the suit number (0-3) for a card."""
    return card // 8


def display_table(table):
    """Display the 4 cards on the table in readable format."""
    cards = [card_to_string(table[i]) if table[i] is not None else "Empty" 
             for i in range(4)]
    print(f"    {cards[0]} {cards[1]}")
    print(f"    {cards[2]} {cards[3]}")


def count_suits(table):
    """Count how many cards of each suit are on the table.
    Returns a dictionary with suit -> list of positions."""
    suit_positions = {0: [], 1: [], 2: [], 3: []}
    
    for pos in range(4):
        if table[pos] is not None:
            suit = get_suit(table[pos])
            suit_positions[suit].append(pos)
    
    return suit_positions


def remove_cards(table, suit_positions):
    """Remove cards according to the game rules.
    Returns True if any cards were removed."""
    
    # Count how many suits have cards
    suits_with_cards = [suit for suit, positions in suit_positions.items() 
                        if len(positions) > 0]
    
    # Find suits with 2 or more cards
    suits_with_matches = [suit for suit in suits_with_cards 
                          if len(suit_positions[suit]) >= 2]
    
    if len(suits_with_matches) == 0:
        # No matches at all
        return False
    
    cards_removed = False
    
    for suit in suits_with_matches:
        positions = suit_positions[suit]
        
        if len(positions) == 4:
            # All 4 cards same suit - remove all 4
            for pos in positions:
                table[pos] = None
            cards_removed = True
        elif len(positions) == 3:
            # 3 cards same suit - remove the 2 in lower positions
            positions.sort()
            table[positions[0]] = None
            table[positions[1]] = None
            cards_removed = True
        elif len(positions) == 2:
            # 2 cards same suit - remove both
            table[positions[0]] = None
            table[positions[1]] = None
            cards_removed = True
    
    return cards_removed


def refill_table(table, deck, deck_index):
    """Refill empty positions from the deck.
    Returns the updated deck_index."""
    for pos in range(4):
        if table[pos] is None and deck_index < len(deck):
            table[pos] = deck[deck_index]
            deck_index += 1
    
    return deck_index


def check_loss(table, deck_index, deck):
    """Check if all 4 suits are present on the table while cards remain in deck."""
    if deck_index >= len(deck):
        # No cards left in deck
        return False
    
    suits_on_table = set()
    for pos in range(4):
        if table[pos] is not None:
            suits_on_table.add(get_suit(table[pos]))
    
    # Loss condition: all 4 suits present while deck has cards
    return len(suits_on_table) == 4


def play_game(seed_value):
    """Play one game of solitaire with the given seed.
    Returns the number of cards left undealt."""
    
    # Create and shuffle deck
    cards = list(range(32))
    random.seed(seed_value)
    random.shuffle(cards)
    
    # Initialize table with first 4 cards
    table = [cards[0], cards[1], cards[2], cards[3]]
    deck_index = 4
    
    # Display initial table
    print()
    display_table(table)
    
    while True:
        # Try to remove cards according to rules
        suit_positions = count_suits(table)
        cards_removed = remove_cards(table, suit_positions)
        
        # If cards were removed, refill and continue
        if cards_removed:
            deck_index = refill_table(table, cards, deck_index)
            
            # Display table after refill
            print()
            display_table(table)
            
            # After refilling, continue to next iteration
            continue
        
        # No cards were removed - check win/lose conditions
        
        # Win condition: deck is empty and no moves left
        if deck_index >= len(cards):
            print("\nWIN")
            return 0
        
        # Loss condition: all 4 suits present with cards still in deck
        if check_loss(table, deck_index, cards):
            print("\nLOSE")
            return len(cards) - deck_index
        
        # This shouldn't happen, but safety check
        print("\nLOSE")
        return len(cards) - deck_index


def simulate(n, i):
    """Simulate n games using seeds i through i+n-1.
    Print percentages of games ending with each number of cards left."""
    
    cards_left_counts = {}
    
    for seed_value in range(i, i + n):
        # Suppress output during simulation
        import sys
        import io
        old_stdout = sys.stdout
        sys.stdout = io.StringIO()
        
        cards_left = play_game(seed_value)
        
        sys.stdout = old_stdout
        
        if cards_left not in cards_left_counts:
            cards_left_counts[cards_left] = 0
        cards_left_counts[cards_left] += 1
    
    # Sort by cards left (ascending)
    sorted_results = sorted(cards_left_counts.items())
    
    # Print percentages (only > 0)
    for cards_left, count in sorted_results:
        percentage = (count / n) * 100
        if percentage > 0:
            print(f"{cards_left} {percentage:.2f}")


def main():
    """Main function for Part 1."""
    seed_input = input("Enter an integer to pass to the seed() function: ")
    seed_value = int(seed_input)
    
    play_game(seed_value)


if __name__ == "__main__":
    main()
