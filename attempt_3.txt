def get_suit(card):
    """Return the suit number (0-3) for a card."""
    return card // 8


def play_game(seed, show_output=True):

    Play a single game of solitaire with the given seed.
    Returns the number of cards left undealt when the game ends.

    # Initialize and shuffle deck

        # Check for exactly 3 cards sharing a suit

            # Find the suit with 3 cards

                if count == 3:
                    suit_with_3 = suit
                    break
            # Find positions of cards with this suit

def simulate(n, i):
    """
    Simulate n games starting from seed i.
    Print statistics about cards left undealt.
    """
    cards_left_counts = {}
    
    for seed in range(i, i + n):
        cards_left = play_game(seed, show_output=False)
        cards_left_counts[cards_left] = cards_left_counts.get(cards_left, 0) + 1
    
    # Print statistics
    for cards_left in sorted(cards_left_counts.keys()):
        percentage = (cards_left_counts[cards_left] / n) * 100
        if percentage > 0:
            print(f"{cards_left} {percentage:.2f}")
